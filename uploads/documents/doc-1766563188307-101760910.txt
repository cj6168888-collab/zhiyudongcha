class CodeGenesis:
    def __init__(self):
        self.client = docker.from_env()
        # [CN] 使用配置了清华源的镜像
        self.image = 
"avatar/runtime:cn_fat" 
        self.logger = logging.getLogger(

)

    def _is_safe_code(self, code: str) -> bool:
        """AST 静态安全扫描"""
        try
:
            tree = ast.parse(code)
            for node in
 ast.walk(tree):
                if isinstance
(node, (ast.Import, ast.ImportFrom)):
                    for alias in
 node.names:
                        if alias.name.split('.')[0] in ['subprocess', 'shutil', 'socket'
]:
                            return False # 拦截高危库
                if isinstance(node, ast.Call) and isinstance
(node.func, ast.Name):
                    if node.func.id in ['eval', 'exec', 'open', 'exit'
]:
                        return False # 拦截高危函数
            return True
        except
 SyntaxError:
            return False

    def generate_and_run(self, prompt: str, llm_brain) -> str:
        # 1. 思考 (通过 llm_brain 调用本地或云端)
        # 注意：这里 llm_brain 是 LocalSoul，它可能需要请求服务端 router
        code = llm_brain.think_code(prompt) 
        
        # 2. 安检
        if not
 self._is_safe_code(code):
            return "Security Alert: Generated code contains dangerous modules."

        # 3. 运行 (Sandbox)
        return
 self._run_in_docker(code)

    def _run_in_docker(self, script: str) -> str:

:

True

import

import

class PCNetwork:
    def __init__(self, server_url, identity, compute_node, async_queue):
        # 将 http 换成 ws
        self.ws_url = server_url.replace(
"http", "ws") + f"/ws/client/{identity['device_fingerprint']}"
        self.token = identity[
'access_token'
]
        self.fingerprint = identity[
'device_fingerprint'
]
        self.compute = compute_node
        self.queue = async_queue
        self.ws = 
None

    def connect(self):
        # 启动队列监听 (用于 Muse 异步回传)
        threading.Thread(target=self._monitor_async_results, daemon=
True
).start()
        
        self.ws = websocket.WebSocketApp(
            self.ws_url,
            on_open=self.on_open,
            on_message=self.on_message
            # on_error, on_close...
        )
        self.ws.run_forever()

    def on_open(self, ws):
        # [Ultimate Fix] 发送符合 Z1 标准的握手
        payload = {
            "type": "handshake_req"
,
            "metadata": {"token"
: self.token},
            "payload"
: {
                "device_fingerprint"
: self.fingerprint,
                "installed_skills": ["coding_v2", "muse_v1"
]
            }
        }
        ws.send(json.dumps(payload))

    def on_message(self, ws, message):
        data = json.loads(message)
        if data.get("type") == "compute_offload"
:
            # 处理服务端下发的计算任务
            def send_back(t, p):
 self.send_packet(t, p)
            self.compute.handle_offload(data.get(
"payload"
), send_back)

    def send_packet(self, msg_type, payload):
        packet = {
            "type"
: msg_type,
            "payload"
: payload,
            "metadata": {"token"
: self.token}
        }
        if
 self.ws: self.ws.send(json.dumps(packet))
        
    def _monitor_async_results(self):
        while True
:
            res = self.queue.get()
            self.send_packet(
"compute_result"
, res)
