 self._call_doubao(prompt)

        except Exception as
 e:

)
            # 3. 熔断降级 (Fallback)
            return
 self._call_tongyi(prompt)

    def _route_policy(self, task_type: str, device_type: str) -> str:
        # 手机端优先省流/速度 -> 豆包
        if device_type == "MOBILE" and task_type == "CHAT"

import
 time

class SpiritCore:
    """
    分形灵魂：根据设备活跃度决定小智此刻在哪个设备上“显灵”。
    """
    def __init__(self):
        self.current_mood = 
"HAPPY"
        # 设备权重: {node_id: {"score": 80, "last_seen": ts, "type": "PC"}}
        self.device_weights = {} 

    def update_weight(self, node_id: str, score: int, node_type: str):
        self.device_weights[node_id] = {
            "score"
: score,
            "last_seen"
: time.time(),
            "type"
: node_type
        }

    def decide_action(self, current_node_id: str, psy_analysis: dict, context: dict) -> dict:
        """
        注意力仲裁：只有 Master Node 能触发灵魂动作
        """
        # 1. 更新当前节点状态
        score = context.get(
"activity_score", 0
)
        node_type = context.get(
"device_type", "MOBILE"
)
        self.update_weight(current_node_id, score, node_type)

        # 2. 选举 Master
        now = time.time()
        active_nodes = {
            k: v 
for k, v in
 self.device_weights.items() 
            if now - v['last_seen'] < 10 # 10s 超时
        }
        
        if not active_nodes: return {"action": "IDLE"
}
        
        master_id = 
max(active_nodes, key=lambda k: active_nodes[k]['score'
])

        # 3. 仲裁
        if
 current_node_id != master_id:
            # 如果我是手机(Mobile)，但主人正在用电脑(PC, master)，
            # 手机应该保持静默，不要随便插话。
            return {"action": "IGNORE", "master"
: master_id}

        # 4. Master 节点的意志
        if psy_analysis.get("has_music"
):
            return
 {
                "action": "DANCE"
, 
                "params": {"bpm": psy_analysis.get("bpm"
)}
            }
            
        return {"action": "IDLE"
}
