import
 androidx.room.*

@Entity(tableName = "node_metadata")
data class NodeMetadata
(
    @PrimaryKey val id: Int = 1
,
    val node_id: String,          // 领养后获得的唯一身份ID
    val user_id: String,          // 归属的 Root 账号
    val sync_hash: String,        // 记忆同步校验码
    val last_dna_update: Long     // 上次同步 Root 基因的时间
)

@Entity(tableName = "offline_packets")
data class OfflinePacket
(
    @PrimaryKey(autoGenerate = true) val id: Long = 0
,
    val
 packet_type: String, 
    val
 payload_json: String,
    
    # [CN-Fractal] 必须存储完整的 context 用于重放
    # 包含: device_id, activity_score, device_type
    val
 metadata_json: String, 
    
    val timestamp: Long
,
    val status: Int = 0 // 0=Pending, 1=Uploading
)

@Dao
interface PacketDao 
{
    @Insert
    fun insert(packet: OfflinePacket)

    # [Safety Fix] 强制分页查询，防止 OOM
    @Query("SELECT * FROM offline_packets WHERE status = 0 ORDER BY timestamp ASC LIMIT :limit")
    fun getPendingPackets(limit: Int = 50)
: List<OfflinePacket>
    
    @Query("DELETE FROM offline_packets WHERE id = :id")
    fun delete(id: Long)

    # 批量确认机制 (ARQ ACK)
    @Query("DELETE FROM offline_packets WHERE id IN (:ids)")
    fun deleteBatch(ids: List<Long>)
}

@Database(entities = [NodeMetadata::class, OfflinePacket::class], version = 1)
abstract class LocalDB : RoomDatabase
() {
    abstract fun packetDao()
: PacketDao
    abstract fun nodeDao(): NodeDao // (省略 NodeDao 实现，逻辑简单)
    
    companion object
 {
        @Volatile private var instance: LocalDB? = null
        fun getInstance(context: android.content.Context)
: LocalDB {
            return instance ?: synchronized(this
) {
                instance ?: Room.databaseBuilder(context, LocalDB::
class.java, "avatar_db"
).build().also { instance = it }
            }
        }
    }
}
