package
 com.avatar.core.logic

import
 android.content.Context
import

import
 android.provider.CalendarContract
import
 org.json.JSONObject

object
 ContextAwareness {
    private var cachedScene: String = "normal"
    private var lastCheckTime: Long = 0

    // [Performance Fix] 增加缓存机制，避免主线程 IO 阻塞
    fun getScene(context: Context)
: String {
        val
 now = System.currentTimeMillis()
        if (now - lastCheckTime > 60000) { // 每分钟更新一次
            cachedScene = queryCalendar(context)
            lastCheckTime = now
        }
        return
 cachedScene
    }
    
    private fun queryCalendar(context: Context)
: String {
        try
 {
            val
 projection = arrayOf(CalendarContract.Instances.BEGIN, CalendarContract.Instances.END)
            val
 now = System.currentTimeMillis()
            val
 cur = context.contentResolver.query(
                CalendarContract.Instances.CONTENT_URI.buildUpon()
                    .appendPath(
"$now"
)
                    .appendPath(
"${now + 3600000}"
)
                    .build(),
                projection, 
null, null, null
            )
            var isMeeting = false
            cur?.use { 
if (it.moveToFirst()) isMeeting = true
 }
            return if (isMeeting) "meeting" else "normal"
        } 
catch
 (e: Exception) {
            return "normal"
        }
    }

    // [Fractal] 实时计算活跃度
    fun getActivityScore(context: Context): Int
 {
        val pm = context.getSystemService(Context.POWER_SERVICE) as
 PowerManager
        return if (pm.isInteractive) 80 else 10
    }

    // 音频环境分析 (DSP)
    fun analyzeAudioEnv(pcmData: ShortArray)
: JSONObject {
        val
 env = JSONObject()
        var sum = 0.0
        for (sample in
 pcmData) sum += sample * sample
        val
 rms = Math.sqrt(sum / pcmData.size)
        env.put(
"volume"
, rms)
        env.put(
"music_playing", rms > 2000
) 
        env.put(
"bpm", if (rms > 2000) 120 else 0
)
        return
 env
    }
}

package com.avatar.core.data

import
 androidx.room.*

@Entity(tableName = "offline_packets")
data class OfflinePacket
(
    @PrimaryKey(autoGenerate = true) val id: Long = 0
,
    val
 packet_type: String, 
    val
 payload_json: String,
    val
 metadata_json: String, 
    val timestamp: Long
,
    val status: Int = 0 
)

@Dao
interface PacketDao 
{
    @Insert
    fun insert(packet: OfflinePacket)

    @Query("SELECT * FROM offline_packets WHERE status = 0 ORDER BY timestamp ASC LIMIT :limit")
    fun getPendingPackets(limit: Int = 50)
: List<OfflinePacket>
    
    @Query("DELETE FROM offline_packets WHERE id IN (:ids)")
    fun deleteBatch(ids: List<Long>)
}

@Database(entities = [OfflinePacket::class], version = 1)
abstract class LocalDB : RoomDatabase
() {
    abstract fun packetDao()
: PacketDao
    
    companion object
 {
        @Volatile private var instance: LocalDB? = null
        fun getInstance(context: android.content.Context)
: LocalDB {
            return instance ?: synchronized(this
) {
                instance ?: Room.databaseBuilder(context, LocalDB::
class.java, "avatar_db"
).build().also { instance = it }
            }
        }
    }
