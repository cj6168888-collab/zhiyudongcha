class DimensionalCore:
    def __init__(self):
        self.identity = "Master"         # 身份标识：总台
        self.privacy_mode = "Stealth"    # 默认隐私模式：幻影
        self.knowledge_base = "Local"    # 优先使用本地法律/财务库
        
    def context_aware_loop(self):
        """核心感知循环：判断谁在场，决定如何交互"""
        presence = self.sensor.detect_voices()
        if presence.only_owner():
            self.interaction.output_channel = ["Speaker", "Screen"]
        else:
            self.interaction.output_channel = ["Bluetooth_Earphone", "Haptic"]
            self.screen.hide_sensitive_info()

    def action_executor(self, task):
        """执行逻辑：如果API不通，启动视觉锚定"""
        try:
            self.os_bridge.execute(task)
        except PermissionError:
            self.visual_driver.click_by_coordinates(task.target_visual)

    def emergency_melt(self):
        """自毁程序：抹除本地，保全云端"""
        self.sync.upload_last_snapshot()
        self.storage.shred_all_local_data()
        self.core.shutdown()
